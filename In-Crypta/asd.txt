# Prompt de Reparación Crítica del Exportador y Certificación Física Headless v0.2.5 (GDevelop 5.6)

## Contexto y Problema Crítico Actual
Trabajás en `~/unahur/PlataformeroMDA` (rama `v0.2.4` rumbo a `v0.2.5`).
Actualmente, al intentar abrir `export/gdevelop_nativo/IN_CRYPTA_NATIVO.json` en GDevelop 5.6, el IDE arroja error indicando que **no es un archivo de GDevelop**.

Las causas identificadas en el repositorio son:
1. **Esquema raíz incompleto tras quitar `BASE_PROJECT` (`commit 61552e9` / `T402`):** `construir_proyecto()` en `tools/generar_proyecto_nativo.py` genera un JSON sin `"gdVersion": {"build": 99, "major": 4, "minor": 0, "revision": 0}`, `"eventsFunctionsExtensions": []`, `"externalEvents": []`, `"externalLayouts": []`, `"externalSourceFiles": []`, `"objectsGroups": []`, `"objectsFolderStructure"` ni `"platforms"` / `"currentPlatform"` en `"properties"`. GDevelop exige estas claves para reconocer el archivo como proyecto válido.
2. **Parche enmascarado en `tools/validar_eventos_gd.py` durante `T302`:** Según `reports/resumen.json`, en `T302` se modificó `_eventos_de` en `validar_eventos_gd.py` para tolerar contenedores de eventos que no fueran listas. En GDevelop, todo `layout["events"]` y todo sub-bloque `"events"` DEBE ser estrictamente una `list` de objetos dict de evento (`BuiltinCommonInstructions::Standard`). Si queda una lista anidada (`[[...]]`) o un dict suelto, GDevelop falla al deserializar.
3. **Descarte de `export/`:** En `T701` se ejecutó `git checkout -- export/`, dejando `export/gdevelop_nativo/IN_CRYPTA_NATIVO.json` desactualizado o corrupto en disco.

## Fase 1 — Simulador y Validador Headless en Node.js (`tests/test_fisica_gdjs.js`)

Creá el archivo `tests/test_fisica_gdjs.js` para probar empíricamente el JSON exportado y simular el comportamiento exacto de `PlatformerObjectRuntimeBehavior` de GDevelop 5.6.285 junto con las reglas de `bloque_gravedad` (`H1`, `H2`, `H3`, `H5` y `T305`):

```javascript
// tests/test_fisica_gdjs.js — Validación de esquema GDevelop 5.6 + Simulación de PlatformBehavior
const fs = require("fs");
const path = require("path");
const assert = require("assert");

const PROJECT_PATH = path.resolve(
  __dirname,
  "..",
  "export",
  "gdevelop_nativo",
  "IN_CRYPTA_NATIVO.json"
);

const project = JSON.parse(fs.readFileSync(PROJECT_PATH, "utf8"));

// 1. Validación estricta del esquema ProjectFile de GDevelop 5.6
assert.strictEqual(typeof project.gdVersion, "object", "Falta gdVersion (objeto)");
assert.strictEqual(typeof project.gdVersion.major, "number", "Falta gdVersion.major");
assert.ok(Array.isArray(project.layouts) && project.layouts.length > 0, "Faltan layouts");
assert.ok(Array.isArray(project.eventsFunctionsExtensions), "Falta eventsFunctionsExtensions");
assert.ok(Array.isArray(project.externalEvents), "Falta externalEvents");
assert.ok(Array.isArray(project.externalLayouts), "Falta externalLayouts");
assert.ok(
  project.properties &&
    Array.isArray(project.properties.platforms) &&
    project.properties.platforms.length > 0,
  "Falta properties.platforms"
);
const layoutNames = new Set(project.layouts.map((l) => l.name));
assert.ok(layoutNames.has(project.firstLayout), `firstLayout '${project.firstLayout}' no existe`);

for (const layout of project.layouts) {
  assert.ok(Array.isArray(layout.events), `layout ${layout.name}: events no es Array`);
  const checkEvents = (events, ctx) => {
    for (let i = 0; i < events.length; i++) {
      const ev = events[i];
      assert.ok(
        ev && typeof ev === "object" && !Array.isArray(ev),
        `${ctx}[${i}] no es un objeto de evento válido`
      );
      assert.strictEqual(
        ev.type,
        "BuiltinCommonInstructions::Standard",
        `${ctx}[${i}] tiene type inválido: ${ev.type}`
      );
      assert.ok(Array.isArray(ev.conditions), `${ctx}[${i}].conditions no es Array`);
      assert.ok(Array.isArray(ev.actions), `${ctx}[${i}].actions no es Array`);
      if (ev.events) {
        assert.ok(Array.isArray(ev.events), `${ctx}[${i}].events no es Array`);
        checkEvents(ev.events, `${ctx}[${i}].events`);
      }
    }
  };
  checkEvents(layout.events, `layout(${layout.name}).events`);
}
console.log("PASS esquema_projectfile_gdevelop_5_6");

// 2. Extraer constantes del PlatformerObject del Jugador en Nivel 1
const nivel1 = project.layouts.find((l) => l.name === "Nivel 1");
assert.ok(nivel1, "No se encontró escena Nivel 1");
const todosObjetos = [...(project.objects || []), ...(nivel1.objects || [])];
const jugadorObj = todosObjetos.find((o) => o.name === "Jugador");
assert.ok(jugadorObj, "No se encontró el objeto Jugador");
const platBehavior = (jugadorObj.behaviors || []).find(
  (b) => b.name === "PlatformerObject" || b.type === "PlatformBehavior::PlatformerObjectBehavior"
);
assert.ok(platBehavior, "Jugador no tiene PlatformerObjectBehavior");

const GRAVITY_BASE = Number(platBehavior.gravity || 1500);
const JUMP_SPEED_BASE = Number(platBehavior.jumpSpeed || 620);
const MAX_FALL_SPEED = Number(platBehavior.maxFallingSpeed || 900);

// 3. Simulador fiel de gdjs.PlatformerObjectRuntimeBehavior._updateFromEvents + preEvents
function simularSalto({ mantenerTecla = true, cortarEnFrame = null, iniciarPostDash = false }) {
  const dt = 1 / 60;
  let y = 0;
  let currentJumpSpeed = iniciarPostDash ? 0 : JUMP_SPEED_BASE;
  let currentFallSpeed = 0;
  let isOnFloor = false;
  let isJumping = !iniciarPostDash;
  let dashing = 0;
  let post_dash = iniciarPostDash ? 0.06 : 0;
  let gravedad_actual = GRAVITY_BASE;
  let minY = 0;
  const historialGravedad = [];

  for (let frame = 0; frame < 90; frame++) {
    const teclaMantenida = mantenerTecla && (cortarEnFrame === null || frame < cortarEnFrame);
    const isFalling = !isOnFloor && !isJumping && currentFallSpeed > 0;

    // AbortJump (H5): en GDJS, abortJump() pone currentJumpSpeed = 0 e isJumping = false
    if (cortarEnFrame !== null && frame === cortarEnFrame && isJumping) {
      currentJumpSpeed = 0;
      isJumping = false;
    }

    // Regla de decisión de gravedad por frame (T302)
    gravedad_actual = GRAVITY_BASE;
    if (!isOnFloor && teclaMantenida && isJumping && currentJumpSpeed < 70) {
      gravedad_actual = Math.floor(GRAVITY_BASE * 0.55); // 825
    }
    if (!isOnFloor && teclaMantenida && isFalling && currentFallSpeed < 70) {
      gravedad_actual = Math.floor(GRAVITY_BASE * 0.55); // 825
    }
    if (post_dash > 0) {
      gravedad_actual = Math.floor(GRAVITY_BASE * 0.5); // 750
    }
    if (dashing === 1) {
      gravedad_actual = 0;
    }
    historialGravedad.push(gravedad_actual);

    // Integración vertical idéntica a platformerobjectruntimebehavior.ts de GDJS 5.6
    const deltaY = (currentFallSpeed - currentJumpSpeed) * dt;
    y += deltaY;
    if (y < minY) minY = y;

    if (isJumping) {
      currentJumpSpeed -= gravedad_actual * dt;
      if (currentJumpSpeed <= 0) {
        currentJumpSpeed = 0;
        isJumping = false;
      }
    } else if (!isOnFloor) {
      currentFallSpeed = Math.min(MAX_FALL_SPEED, currentFallSpeed + gravedad_actual * dt);
    }

    if (post_dash > 0) {
      post_dash = Math.max(0, post_dash - dt);
    }

    if (frame > 0 && y >= 0) {
      y = 0;
      isOnFloor = true;
      currentFallSpeed = 0;
      // Frame posterior al aterrizaje: verificar que la gravedad regrese a 1500
      gravedad_actual = GRAVITY_BASE;
      historialGravedad.push(gravedad_actual);
      break;
    }
  }

  return { alturaMax: -minY, historialGravedad };
}

// Prueba H1 + H3: Salto sin mantener tecla (sin ápice) -> altura ~128 px y gravedad constante 1500
const saltoSinApice = simularSalto({ mantenerTecla: false });
assert.ok(
  saltoSinApice.alturaMax >= 120 && saltoSinApice.alturaMax <= 136,
  `Altura sin ápice fuera de rango: ${saltoSinApice.alturaMax.toFixed(1)} px (esperado ~128 px)`
);
assert.ok(
  saltoSinApice.historialGravedad.every((g) => g === 1500),
  "Sin tecla mantenida la gravedad nunca debe bajar de 1500"
);
console.log(`PASS H1_salto_sin_apice (altura=${saltoSinApice.alturaMax.toFixed(1)}px)`);

// Prueba H1 + H3: Salto manteniendo tecla (con ápice) -> activa 825 solo en cima y vuelve a 1500
const saltoConApice = simularSalto({ mantenerTecla: true });
assert.strictEqual(saltoConApice.historialGravedad[0], 1500, "En despegue la gravedad debe ser 1500");
assert.ok(saltoConApice.historialGravedad.includes(825), "En el ápice la gravedad debe bajar a 825");
assert.strictEqual(
  saltoConApice.historialGravedad[saltoConApice.historialGravedad.length - 1],
  1500,
  "Al aterrizar la gravedad debe volver a 1500"
);
assert.ok(
  saltoConApice.alturaMax >= 124 && saltoConApice.alturaMax <= 140,
  `Altura con ápice fuera de rango: ${saltoConApice.alturaMax.toFixed(1)} px`
);
console.log(`PASS H1_H3_salto_con_apice_y_recuperacion (altura=${saltoConApice.alturaMax.toFixed(1)}px)`);

// Prueba H2: Post-dash float -> empieza en 750 durante ~0.06s y vuelve a 1500
const simPostDash = simularSalto({ mantenerTecla: false, iniciarPostDash: true });
assert.strictEqual(simPostDash.historialGravedad[0], 750, "Post-dash debe iniciar con gravedad 750");
assert.strictEqual(simPostDash.historialGravedad[5], 1500, "Tras 0.06s (4 frames) debe volver a 1500");
console.log("PASS H2_post_dash_float");

// Prueba H5: Salto corto (AbortJump al frame 6 = 0.10s)
const saltoCorto = simularSalto({ mantenerTecla: true, cortarEnFrame: 6 });
assert.ok(
  saltoCorto.alturaMax >= 45 && saltoCorto.alturaMax <= 75,
  `Salto corto fuera de rango: ${saltoCorto.alturaMax.toFixed(1)} px`
);
console.log(`PASS H5_corte_salto (altura=${saltoCorto.alturaMax.toFixed(1)}px)`);

console.log("TESTS_FISICA_GDJS_OK");
```

## Fase 2 — Cerrar `T305` y los Hallazgos `I1`–`I8`

1. **Tarea T305 (Umbral de Pisotón al Jefe):**
   - Tanto `Jugador` (`32x32`, máscara `24x30` que llega hasta `y + 32`) como `Jefe` (`48x64`) tienen `centerPoint`/`origin` en la esquina superior izquierda `(0, 0)`.
